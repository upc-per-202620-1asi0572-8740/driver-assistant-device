La aplicación Kärcher Home Robots permitirá vincular la estación MopCare M1 y consultar desde un teléfono o tablet el nivel de agua limpia disponible. La pantalla principal mostrará el porcentaje del depósito, su capacidad aproximada en mililitros y una alerta cuando sea necesario reponer agua.

El servicio de preparación de la bandeja mostrará si esta se encuentra cerca de la estación, el peso medido y el estado de la válvula. La dosificación comenzará cuando la bandeja esté a 15 cm o menos, exista agua y el peso sea inferior a 200 g; finalizará al alcanzar 800 g. La interfaz indicará cuándo es seguro retirar la bandeja y continuar el enjuague del paño.

El servicio de avisos notificará al usuario si el depósito está vacío o si se interrumpe un ciclo por falta de agua. En el equipo, la pantalla LCD mostrará el nivel del depósito y el peso de la bandeja; cuando no quede agua aparecerá «Empty» y la válvula permanecerá cerrada hasta que se reponga el depósito.

Para soporte técnico se propone una interfaz web o desktop con el historial de ciclos, eventos de proximidad, cambios de modo y peso medido al terminar cada dosificación. Cada registro distinguirá los ciclos completados de los interrumpidos, de modo que soporte pueda comprobar si la válvula se cerró al alcanzar el peso previsto. El prototipo de Wokwi representará este seguimiento en la consola serial, sin implementar la aplicación móvil ni el servicio remoto.

MopCareDevice.h

#ifndef MOP_CARE_DEVICE_H
#define MOP_CARE_DEVICE_H

/**
 * @file MopCareDevice.h
 * @brief Sensor and valve controller for MopCare M1.
 * @author sebastian rodriguez
 */

#include <Arduino.h>
#include <ModestIoT.h>
#include <math.h>

// Pin definitions
static const int RESERVOIR_TRIG_PIN = 4;
static const int RESERVOIR_ECHO_PIN = 5;
static const int PROXIMITY_TRIG_PIN = 16;
static const int PROXIMITY_ECHO_PIN = 17;
static const int SCALE_DATA_PIN = 32;
static const int SCALE_CLOCK_PIN = 33;
static const int VALVE_SERVO_PIN = 19;
static const uint8_t LCD_ADDRESS = 0x27;

// Constants
static const float FULL_DISTANCE_CM = 5.0f;
static const float EMPTY_DISTANCE_CM = 35.0f;
static const float TRAY_DISTANCE_CM = 15.0f;
static const float RESERVOIR_CAPACITY_ML = 3000.0f;
static const float SCALE_CAPACITY_GRAMS = 5000.0f;
static const float TRAY_CAPACITY_GRAMS = 1000.0f;
static const float START_WEIGHT_GRAMS = 200.0f;
static const float TARGET_WEIGHT_GRAMS = 800.0f;
static const long SCALE_MIN_RAW = 0;
static const long SCALE_MAX_RAW = 2100;
static const int FILTER_DEPTH = 5;
static const int LCD_COLUMNS = 16;
static const int LCD_ROWS = 2;
static const int SERVO_CLOSED_ANGLE = 0;
static const unsigned long SAMPLING_INTERVAL_MS = 100;
static const uint8_t QUEUE_DEPTH = 16;
static const int TIMER_INDEX = 0;

// Device class
class MopCareDevice : public Device {
private:
    // Sensors
    UltrasonicSensor reservoirSensor;
    UltrasonicSensor proximitySensor;
    LoadCellAmplifier trayScale;

    // Actuators
    ServoMotor valveServo;
    CharacterLcdDisplay lcd;

    // Internal state
    enum class Mode { Waiting, Dosing, Completed, Empty, Interrupted };
    Mode mode;
    int reservoirLevel;
    float trayWeight;
    float lastCycleWeight;
    bool reservoirMeasured;
    bool proximityMeasured;
    bool weightMeasured;
    bool trayNear;
    bool cycleArmed;
    bool refillBlocked;
    unsigned long startedCycles;
    unsigned long completedCycles;
    unsigned long interruptedCycles;

    int calculateLevel(float distance) {
        float percent = (EMPTY_DISTANCE_CM - distance) * 100.0f /
                        (EMPTY_DISTANCE_CM - FULL_DISTANCE_CM);
        return constrain((int)(percent + 0.5f), 0, 100);
    }

    void updateDisplay() {
        char line1[LCD_COLUMNS + 1];
        char line2[LCD_COLUMNS + 1];
        if (reservoirMeasured && reservoirLevel == 0) {
            snprintf(line1, sizeof(line1), "Empty");
        } else {
            snprintf(line1, sizeof(line1), "Agua: %d%%", reservoirLevel);
        }
        snprintf(line2, sizeof(line2), "Bandeja: %.0f g", trayWeight);
        lcd.setLineBuffer(0, line1);
        lcd.setLineBuffer(1, line2);
        lcd.handle(CharacterLcdDisplay::UPDATE_TEXT_COMMAND);
    }

    void setMode(Mode nextMode) {
        if (mode == nextMode) return;
        mode = nextMode;
        const char* name = "Espera";
        switch (mode) {
            case Mode::Waiting: name = "Espera"; break;
            case Mode::Dosing: name = "Dosificando"; break;
            case Mode::Completed: name = "Completado"; break;
            case Mode::Empty: name = "Sin agua"; break;
            case Mode::Interrupted: name = "Interrumpido"; break;
        }
        Serial.printf("Modo: %s\n", name);
    }

    void finishCycle(const char* result, bool completed) {
        valveServo.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);
        lastCycleWeight = trayWeight;
        if (completed) {
            completedCycles++;
            setMode(Mode::Completed);
        } else {
            interruptedCycles++;
            setMode(Mode::Interrupted);
        }
        Serial.printf("Ciclo %lu: %s | Peso final: %.0f g\n",
                      startedCycles, result, lastCycleWeight);
        Serial.printf("Iniciados: %lu | Completados: %lu | Interrumpidos: %lu\n",
                      startedCycles, completedCycles, interruptedCycles);
        if (completed) Serial.println("Bandeja lista para retirar.");
    }

    void controlDosing() {
        if (!reservoirMeasured) return;
        if (reservoirLevel == 0) {
            if (mode == Mode::Dosing) finishCycle("Interrumpido: sin agua", false);
            refillBlocked = true;
            setMode(Mode::Empty);
            return;
        }
        if (refillBlocked) {
            refillBlocked = false;
            Serial.println("Deposito repuesto. Valvula desbloqueada.");
            if (mode == Mode::Empty) setMode(Mode::Waiting);
        }
        if (!proximityMeasured || !weightMeasured) return;

        if (!trayNear) {
            if (mode == Mode::Dosing) finishCycle("Interrumpido: bandeja retirada", false);
            cycleArmed = true;
            if (mode != Mode::Waiting) setMode(Mode::Waiting);
            return;
        }
        if (mode == Mode::Dosing) {
            if (trayWeight >= TARGET_WEIGHT_GRAMS) {
                finishCycle("Completado", true);
            }
            return;
        }
        if (cycleArmed && trayWeight < START_WEIGHT_GRAMS) {
            valveServo.handle(ServoMotor::LOCK_TO_MAXIMUM_LIMIT_COMMAND);
            startedCycles++;
            cycleArmed = false;
            setMode(Mode::Dosing);
            Serial.printf("Inicia ciclo %lu | Peso: %.0f g\n", startedCycles, trayWeight);
        }
    }

public:
    MopCareDevice()
        : Device(SAMPLING_INTERVAL_MS, TIMER_INDEX),
          reservoirSensor(RESERVOIR_TRIG_PIN, RESERVOIR_ECHO_PIN, this),
          proximitySensor(PROXIMITY_TRIG_PIN, PROXIMITY_ECHO_PIN, this),
          trayScale(SCALE_DATA_PIN, SCALE_CLOCK_PIN,
                    0.0f, SCALE_CAPACITY_GRAMS,
                    SCALE_MIN_RAW, SCALE_MAX_RAW, FILTER_DEPTH, this),
          valveServo(VALVE_SERVO_PIN, SERVO_CLOSED_ANGLE, this),
          lcd(LCD_ADDRESS, LCD_COLUMNS, LCD_ROWS, true, this),
          mode(Mode::Waiting),
          reservoirLevel(0),
          trayWeight(0.0f),
          lastCycleWeight(0.0f),
          reservoirMeasured(false),
          proximityMeasured(false),
          weightMeasured(false),
          trayNear(false),
          cycleArmed(true),
          refillBlocked(false),
          startedCycles(0),
          completedCycles(0),
          interruptedCycles(0) {
        updateDisplay();
        trayScale.tare();
        valveServo.handle(ServoMotor::LOCK_TO_MINIMUM_LIMIT_COMMAND);

        appendSensorToScheduler(&reservoirSensor,
                                Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendSensorToScheduler(&proximitySensor,
                                Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER);
        appendToBothSchedulers(&trayScale,
                               Sensor::MEASURE_DATA_REQUESTED_EVENT_IDENTIFIER, CHANGE);
        initializeAsynchronousEngine(QUEUE_DEPTH);
    }

    void on(Event event) override {
        if (event.identifier != Sensor::DATA_READ_EVENT_IDENTIFIER) return;

        if (event.sourceId == RESERVOIR_TRIG_PIN) {
            float distance = reservoirSensor.getDistanceInCentimeters();
            if (distance <= 0.0f) return;
            int currentLevel = calculateLevel(distance);
            if (!reservoirMeasured || currentLevel != reservoirLevel) {
                float waterMl = RESERVOIR_CAPACITY_ML * currentLevel / 100.0f;
                Serial.printf("Deposito: %d%% | %.0f mL\n", currentLevel, waterMl);
            }
            reservoirLevel = currentLevel;
            reservoirMeasured = true;
        } else if (event.sourceId == PROXIMITY_TRIG_PIN) {
            float distance = proximitySensor.getDistanceInCentimeters();
            if (distance <= 0.0f) return;
            bool currentNear = distance <= TRAY_DISTANCE_CM;
            if (!proximityMeasured || currentNear != trayNear) {
                Serial.println(currentNear ? "Bandeja cerca." : "Bandeja retirada.");
            }
            trayNear = currentNear;
            proximityMeasured = true;
        } else if (event.sourceId == SCALE_DATA_PIN) {
            float currentWeight = constrain(trayScale.getWeightInGrams(),
                                            0.0f, TRAY_CAPACITY_GRAMS);
            if (!weightMeasured || fabsf(currentWeight - trayWeight) >= 1.0f) {
                Serial.printf("Peso bandeja: %.0f g\n", currentWeight);
            }
            trayWeight = currentWeight;
            weightMeasured = true;
        } else {
            return;
        }

        controlDosing();
        updateDisplay();
    }
};

#endif
